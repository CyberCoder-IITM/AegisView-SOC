  async def enrich_before_narration(anomaly_dict: dict) -> dict:
    src_ip = anomaly_dict.get("src_ip", "")
    
    # Check darkweb correlator (already built in Phase 3)
    from geo.darkweb_correlator import correlate_ip
    correlation = correlate_ip(src_ip)
    anomaly_dict["darkweb_correlation"] = correlation.dict()
    
    # Check threat intel enrichment
    from geo.threat_intel import enrich_ip
    intel = enrich_ip(src_ip, "")
    anomaly_dict["threat_intel"] = intel.dict()
    
    # Add a verified_context summary for Gemini to use:
    if correlation.threat_category != "CLEAN":
      anomaly_dict["verified_context"] = (
        f"VERIFIED: This IP appears in {correlation.intel_source} "
        f"as category {correlation.threat_category}. "
        f"Feeds matched: {', '.join(correlation.feeds_matched)}."
      )
    elif intel.is_tor:
      anomaly_dict["verified_context"] = (
        "VERIFIED: This IP is a confirmed Tor exit node "
        "per TorProject.org bulk exit list."
      )
    else:
      anomaly_dict["verified_context"] = (
        "UNVERIFIED: This IP has no matches in current "
        "threat intelligence feeds. Treat as unknown."
      )
    
    return anomaly_dict

  Call enrich_before_narration() before generate_narrative()
  Pass enriched dict to Gemini so it has verified context.
